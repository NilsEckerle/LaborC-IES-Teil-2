17static void ENGINE_init_pwm() {
18 // Disable all interrupts
19 cli();
20 // Set prescaler to 64, cf. datasheet for TCCR0B
21 // (TCCR0B: Timer/Counter Control Register 0 B)
22 TCCR0B = 0;
23 SET_BIT(TCCR0B, CS00);
24 SET_BIT(TCCR0B, CS01);
25 // Set waveform generation mode to Fast PWM, frequency = F_CPU / (PRESCALER *
26 // 2^8)
27 TCCR0A = 0;
28 SET_BIT(TCCR0A, WGM00);
29 SET_BIT(TCCR0A, WGM01);
30 // Re-enable all interrupts
31 sei();
32}
