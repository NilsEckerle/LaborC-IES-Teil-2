Adds data by value to the dynamic array.

Adds data by value to the dynamic arrayThis macro creates a heap-allocated copy of the provided data and adds it to the dynamic array. The original data can be safely modified or freed after this operation.

Parameters
tp_arrayPointer to the dynamic array structure
t_dataThe data value to be copied and added (any type)
Note
This macro uses typeof() which is a GCC extension. It allocates memory for a copy of the data, so the destructor will properly free this memory.

int value = 42; DYN_ARR_add_by_value(array, value); // Copies value to heap

#define DYNAMIC_ARRAY_H
#include <stdint.h>
typedef struct dynamic_array {
void **vpp_data_array;
uint8_t ui8_size;
} t_dyn_arr;
t_dyn_arr *DYN_ARRAY_constructor();
int8_t DYN_ARR_add(t_dyn_arr *tp_array, void *vp_data);

int8_t DYN_ARR_destructor(t_dyn_arr **tpp_array);
#define DYN_ARR_add_by_value(tp_array, t_data) do { \
void *vp_data = malloc(sizeof(typeof(t_data))); \
typeof(t_data) tmp = (t_data); \
memcpy(vp_data, &tmp, sizeof(typeof(t_data))); \
DYN_ARR_add(tp_array, vp_data); \
} while (0)
