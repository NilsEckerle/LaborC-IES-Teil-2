22
23 // Reset Timer1 control registers
24 TCCR1A = 0;
25 TCCR1B = 0;
26 TCNT1 = 0;
27
28 // Configure Timer1 for Normal mode (WGM13:0 = 0000)
29 UNSET_BIT(TCCR1A, CLOCK_WGM10_BIT);
30 UNSET_BIT(TCCR1A, CLOCK_WGM11_BIT);
31 UNSET_BIT(TCCR1B, CLOCK_WGM12_BIT);
32 UNSET_BIT(TCCR1B, CLOCK_WGM13_BIT);
33
34 // Set prescaler to 64 (CS12:0 = 011)
35 // This gives us: 16MHz / 8 = 2MHz timer frequency
36 // Timer period = 1/2MHz = 5e-7 microseconds per tick
37 // Overflow occurs every 65536 ticks = 65536 * 5e-7 = 32,768 microseconds
38 UNSET_BIT(TCCR1B, CLOCK_CS10_BIT);
39 SET_BIT(TCCR1B, CLOCK_CS11_BIT);
40 UNSET_BIT(TCCR1B, CLOCK_CS12_BIT);
41
42 // Enable Timer1 overflow interrupt
43 SET_BIT(TIMSK1, CLOCK_TOIE1_BIT);
44
45 // Enable global interrupts
46 sei();
47
48 INFO("[CLOCK_init] Clock initialization successful\n");
49}
