68void STATE_MACHINE_update(t_state_machine *tp_state_machine) {
69 t_state *tp_state = tp_state_machine->tp_current_state;
70
71 INFO_SPAM("[run] running state adress: %p\n", tp_state);
72
73 if (tp_state_machine->bool_is_new_state) {
74 tp_state_machine->bool_is_new_state = 0;
75 tp_state_machine->tp_current_state->on_entry(tp_state_machine->tp_current_state);
76 }
77 if (tp_state->tp_parent != NULL) {
78 tp_state->tp_parent->on_update(tp_state->tp_parent);
79 }
80 tp_state->on_update(tp_state);
81 STATE_check_edges(tp_state, tp_state_machine);
82}
